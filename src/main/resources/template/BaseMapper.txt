import org.apache.ibatis.annotations.Param;
import java.util.List;

public interface BaseMapper<T, P> {

    /**
     * selectList: Query data list based on parameters
     */
    List<T> selectList(@Param("query") P p);

    /**
     * selectCount: Query data count based on parameters
     */
    Integer selectCount(@Param("query") P p);

    /**
     * insert: Insert a new record
     */
    Integer insert(@Param("bean") T t);

    /**
     * insertOrUpdate: Insert or update a record
     */
    Integer insertOrUpdate(@Param("bean") T t);

    /**
     * insertBatch: Batch insert records
     */
    Integer insertBatch(@Param("list") List<T> list);

    /**
     * insertOrUpdateBatch: Batch insert or update records
     */
    Integer insertOrUpdateBatch(@Param("list") List<T> list);

    /**
     * updateByParam: Update records based on multiple conditions
     */
    Integer updateByParam(@Param("bean") T t, @Param("query") P p);

    /**
     * deleteByParam: Delete records based on multiple conditions
     */
    Integer deleteByParam(@Param("query") P p);
}
