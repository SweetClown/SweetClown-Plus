import org.apache.ibatis.annotations.Param;
import java.util.List;

public interface BaseMapper<T, P> {

    /**
     * selectList:(根據參數查詢資料列表)
     */
    List<T> selectList(@Param("query") P p);

    /**
     * selectCount:(根據參數查詢資料筆數)
     */
    Long selectCount(@Param("query") P p);

    /**
     * insert:(新增)
     */
    Long insert(@Param("bean") T t);

    /**
     * insertOrUpdate:(新增或更新)
     */
    Long insertOrUpdate(@Param("bean") T t);

    /**
     * insertBatch:(批次新增)
     */
    Long insertBatch(@Param("list") List<T> list);

    /**
     * insertOrUpdateBatch:(批次新增或更新)
     */
    Long insertOrUpdateBatch(@Param("list") List<T> list);

    /**
     * updateByParams:(多條件更新)
     */
    Long updateByParam(@Param("bean") T t, @Param("query") P p);

    /**
     * deleteByParam:(多條件刪除)
     */
    Long deleteByParam(@Param("query") P p);
}
